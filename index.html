function createCard(bundle) { const card = document.createElement("article"); card.className = "card"; const title = document.createElement("strong"); title.className = "card-title"; setText(title, () => bundle.description || bundle.name); const details = document.createElement("p"); const amount = Number(bundle.dataAmount); const allowance = () => bundle.unlimited ? t("بيانات غير محدودة") : Number.isFinite(amount) && amount > 0 ? t("{amount} جيجابايت", { amount: amount / 1000 }) : t("راجع تفاصيل الباقة"); const unit = () => bundle.durationUnit === "day" ? t("يوم") : bundle.durationUnit || ""; setText(details, () => t("{allowance} — المدة: {duration} {unit}", { allowance: allowance(), duration: bundle.duration ?? t("غير محددة"), unit: unit() })); card.append(title, details); const planIdentity = String(bundle.name || "") + " " + String(bundle.description || ""); const planType = /(?:^|[^a-z])(?:esim_ule_|unlimited\s+essential)/i.test(planIdentity) ? "essential" : /(?:^|[^a-z])(?:esim_ulp_|unlimited\s+plus)/i.test(planIdentity) ? "plus" : null; if (bundle.unlimited === true && planType) { const info = document.createElement("p"); info.className = "notice"; const descriptions = { ar: { essential: "1 جيجابايت بسرعة كاملة لكل دورة يومية، ثم تستمر الخدمة بسرعة 1.25 ميجابت/ثانية. يختلف وقت التجدد حسب نوع الشريحة.", plus: "2 جيجابايت بسرعة كاملة لكل دورة يومية، ثم تستمر الخدمة بسرعة 2 ميجابت/ثانية. يختلف وقت التجدد حسب نوع الشريحة." }, en: { essential: "1 GB at full speed per daily cycle, then unlimited data at 1.25 Mbps. Reset time depends on the eSIM profile.", plus: "2 GB at full speed per daily cycle, then unlimited data at 2 Mbps. Reset time depends on the eSIM profile." }, tr: { essential: "Günlük dönemde 1 GB tam hız, ardından 1,25 Mbps hızında sınırsız veri. Yenilenme zamanı eSIM profiline bağlıdır.", plus: "Günlük dönemde 2 GB tam hız, ardından 2 Mbps hızında sınırsız veri. Yenilenme zamanı eSIM profiline bağlıdır." }, ur: { essential: "ہر روز 1 جی بی مکمل رفتار پر، پھر 1.25 Mbps پر لامحدود ڈیٹا۔ تجدید کا وقت eSIM پروفائل پر منحصر ہے۔", plus: "ہر روز 2 جی بی مکمل رفتار پر، پھر 2 Mbps پر لامحدود ڈیٹا۔ تجدید کا وقت eSIM پروفائل پر منحصر ہے۔" } }; setText(info, () => descriptions[lang][planType]); card.append(info); } const available = bundle.currency === "SAR" && validAmount(bundle.salePrice); const price = document.createElement("p"); price.className = "price"; setText(price, () => available ? t("السعر: {amount} ريال سعودي", { amount: bundle.salePrice.toFixed(2) }) : t("السعر غير متاح حاليًا")); card.append(price); const coveredCountries = Array.isArray(bundle.countries) ? bundle.countries.filter(item => item && (item.iso || item.name)) : []; if (coveredCountries.length) { const coverage = document.createElement("details"); const heading = document.createElement("summary"); const names = document.createElement("p"); const coverageLabels = { ar: "الدول المشمولة بالتغطية", en: "Covered countries", tr: "Kapsanan ülkeler", ur: "کوریج والے ممالک" }; setText(heading, () => coverageLabels[lang] || coverageLabels.ar); setText(names, () => { const labels = [...new Set(coveredCountries.map(arabicName))]; return labels.join(lang === "ar" || lang === "ur" ? "، " : ", "); }); coverage.append(heading, names); card.append(coverage); } const button = document.createElement("button"); button.type = "button"; button.className = "btn"; setText(button, () => t("اختيار الباقة")); button.disabled = !available; const status = document.createElement("p"); status.className = "status"; status.setAttribute("aria-live", "polite"); button.onclick = () => selectBundle(bundle, button, status); card.append(button, status); return card; }
